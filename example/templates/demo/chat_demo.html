JavaScript API Examples

Adding Messages Programmatically

// Get chat instance
const chat = Metro.getPlugin('#myChat', 'chat');

// Add a single message
chat.add({
    name: 'User Name',
    avatar: '👨',
    text: 'Hello, world!',
    position: 'left',
    time: new Date()
});

// Add multiple messages
chat.addMessages([
    {
        name: 'User 1',
        avatar: '👨',
        text: 'Hello!',
        position: 'left',
        time: new Date()
    },
    {
        name: 'User 2',
        avatar: '👩',
        text: 'Hi there!',
        position: 'right',
        time: new Date()
    }
]);

// Delete a message
chat.delMessage('message-id');

// Update a message
chat.updMessage({
    id: 'message-id',
    text: 'Updated message text',
    time: new Date()
});

// Clear all messages
chat.clear();

// Toggle read-only mode
chat.toggleReadonly(true);
