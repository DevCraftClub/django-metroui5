JavaScript API Examples

Programmatic Control

// Add cloak to an element
element.classList.add('cloak');

// Reveal a cloaked element
element.classList.add('remove-cloak');

// Hide a visible element
element.classList.add('add-cloak');

// Remove cloak classes
element.classList.remove('cloak', 'remove-cloak', 'add-cloak');

// Custom animation duration
element.style.animationDuration = '2s';
element.classList.add('remove-cloak');

// Custom animation delay
element.style.animationDelay = '1s';
element.classList.add('remove-cloak');
