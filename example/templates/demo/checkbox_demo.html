JavaScript API Examples

Programmatic Control

{% metroui5_checkbox id="demo-checkbox" caption="Demo Checkbox" %} 
// Get checkbox instance
const checkbox = Metro.getPlugin('#demo-checkbox', 'checkbox');

// Check the checkbox
checkbox.check();

// Uncheck the checkbox
checkbox.uncheck();

// Set to indeterminate state (for three-state checkboxes)
checkbox.indeterminate();

// Toggle the state
checkbox.toggle();

// Get the current state
const state = checkbox.getCheckState();
const stateAsString = checkbox.getCheckState(true);

// Set specific state (-1/unchecked, 0/indeterminate, 1/checked)
checkbox.setCheckState(1);
